% ==========================================================
% 폰트 설정
% ==========================================================
%
% 국문 작성지침은 신명조 / 명조 / 휴먼명조 계열을 요구합니다.
% 이 파일은 지침 서체를 먼저 찾고, 없으면 명조 계열 대체 폰트를
% 순서대로 적용하도록 구성합니다.

\defaultfontfeatures{Ligatures=TeX}

% ----------------------------------------------------------
% 영문/숫자 폰트
% ----------------------------------------------------------
% 표지와 영문초록의 라틴 문자는 Times 계열로 맞춥니다.
\IfFileExists{./fonts/times.ttf}{%
  \setmainfont{Times New Roman}[
    Path=./fonts/,
    Extension=.ttf,
    UprightFont=times,
    BoldFont=timesbd,
    ItalicFont=timesi,
    BoldItalicFont=timesbi
  ]%
}{%
  \IfFontExistsTF{Times New Roman}{%
    \setmainfont{Times New Roman}%
  }{%
    \setmainfont{TeX Gyre Termes}%
    \PackageWarningNoLine{gachon-thesis}{Times New Roman was not found. Falling back to TeX Gyre Termes for Latin text}%
  }%
}

% ----------------------------------------------------------
% 한글 본문 폰트
% ----------------------------------------------------------
% Priority:
%   1) Uploaded HCR Batang files in ./fonts/
%   2) System-installed guideline serif fonts
%   3) Other Myeongjo-style fonts commonly available
\newcommand{\GachonKRSetHangulFontFallback}{%
  \IfFontExistsTF{HY신명조}{%
    \setmainhangulfont{HY신명조}%
  }{%
    \IfFontExistsTF{HYSMyeongJo}{%
      \setmainhangulfont{HYSMyeongJo}%
    }{%
      \IfFontExistsTF{HYShinMyeongJo-Medium}{%
        \setmainhangulfont{HYShinMyeongJo-Medium}%
      }{%
        \IfFontExistsTF{AppleMyungjo}{%
          \setmainhangulfont{AppleMyungjo}%
        }{%
          \IfFontExistsTF{Batang}{%
            \setmainhangulfont{Batang}%
          }{%
            \IfFontExistsTF{HCR Batang}{%
              \setmainhangulfont{HCR Batang}%
            }{%
              \IfFontExistsTF{NanumMyeongjo}{%
                \setmainhangulfont{NanumMyeongjo}[BoldFont={NanumMyeongjo ExtraBold}]%
              }{%
                \PackageWarningNoLine{gachon-thesis}{No guideline-serif Hangul font was found. kotex default font will be used. Install or upload a Myeongjo-family font for submission}%
              }%
            }%
          }%
        }%
      }%
    }%
  }%
}

\IfFileExists{./fonts/HANBatang.ttf}{%
  \setmainhangulfont{HCR Batang}[
    Path=./fonts/,
    Extension=.ttf,
    UprightFont=HANBatang,
    BoldFont=HANBatangB
  ]%
}{%
  \GachonKRSetHangulFontFallback
}
